app.config['UPLOAD_FOLDER'] = os.environ.get('UPLOAD_FOLDER', 'uploads')

# Initialize database
def init_db():
    conn = sqlite3.connect(app.config['DATABASE'])
    c = conn.cursor()
    c.execute('''CREATE TABLE IF NOT EXISTS users
                 (id INTEGER PRIMARY KEY AUTOINCREMENT,
                 username TEXT UNIQUE NOT NULL,
                 password TEXT NOT NULL,
                 email TEXT,
                 created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP)''')
    
    c.execute('''CREATE TABLE IF NOT EXISTS torrents
                 (id INTEGER PRIMARY KEY AUTOINCREMENT,
                 title TEXT NOT NULL,
                 description TEXT,
                 filename TEXT NOT NULL,
                 encrypted_filename TEXT NOT NULL,
                 size INTEGER NOT NULL,
                 uploader_id INTEGER NOT NULL,
                 created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                 decryption_key TEXT NOT NULL,
                 FOREIGN KEY(uploader_id) REFERENCES users(id))''')
    conn.commit()
    conn.close()

init_db()

# Encryption function
def encrypt_file(input_path, output_path, password):
    salt = get_random_bytes(16)
    key = hashlib.pbkdf2_hmac('sha256', password.encode('utf-8'), salt, 100000, dklen=32)
    iv = get_random_bytes(16)
    cipher = AES.new(key, AES.MODE_CBC, iv)
    file_size = os.path.getsize(input_path)
    
    with open(output_path, 'wb') as fout:
        fout.write(salt)
        fout.write(iv)
        fout.write(file_size.to_bytes(8, 'big'))
        
        with open(input_path, 'rb') as fin:
            while True:
                chunk = fin.read(1024 * 1024)  # 1MB chunks
                if len(chunk) == 0:
                    break
                elif len(chunk) % 16 != 0:
                    chunk = pad(chunk, AES.block_size)
                fout.write(cipher.encrypt(chunk))

# Decryption function
def decrypt_file(input_path, output_path, password):
    with open(input_path, 'rb') as fin:
        salt = fin.read(16)
        iv = fin.read(16)
        file_size = int.from_bytes(fin.read(8), 'big')
        key = hashlib.pbkdf2_hmac('sha256', password.encode('utf-8'), salt, 100000, dklen=32)
        cipher = AES.new(key, AES.MODE_CBC, iv)
        
        with open(output_path, 'wb') as fout:
            remaining = file_size
            while True:
                chunk = fin.read(1024 * 1024)
                if len(chunk) == 0:
                    break
                    
                decrypted = cipher.decrypt(chunk)
                
                if remaining < len(decrypted):
                    decrypted = decrypted[:remaining]
                    remaining = 0
                else:
                    remaining -= len(decrypted)
                
                fout.write(decrypted)
                if remaining <= 0:
                    break

# Database helper
def get_db():
    conn = sqlite3.connect(app.config['DATABASE'])
    conn.row_factory = sqlite3.Row
    return conn

@app.route('/')
def index():
    conn = get_db()
    cursor = conn.cursor()
    cursor.execute('''SELECT torrents.*, users.username 
                   FROM torrents 
                   JOIN users ON torrents.uploader_id = users.id 
                   ORDER BY torrents.created_at DESC''')
    torrents = cursor.fetchall()
    conn.close()
    return render_template('index.html', torrents=torrents)

@app.route('/register', methods=['GET', 'POST'])
def register():
    if request.method == 'POST':
        username = request.form['username']
        password = request.form['password']
        email = request.form.get('email', '')
        
        if not username or not password:
            flash('Username and password are required', 'danger')
            return redirect(url_for('register'))
        
        hashed_password = generate_password_hash(password)
        
        try:
            conn = get_db()
            cursor = conn.cursor()
            cursor.execute('INSERT INTO users (username, password, email) VALUES (?, ?, ?)', 
                           (username, hashed_password, email))
            conn.commit()
            flash('Account created! Please log in.', 'success')
            return redirect(url_for('login'))
        except sqlite3.IntegrityError:
            flash('Username already exists', 'danger')
        finally:
            conn.close()
    
    return render_template('register.html')

@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        username = request.form['username']
        password = request.form['password']
        
        conn = get_db()
        cursor = conn.cursor()
        cursor.execute('SELECT * FROM users WHERE username = ?', (username,))
        user = cursor.fetchone()
        conn.close()
        
        if user and check_password_hash(user['password'], password):
            session['user_id'] = user['id']
            session['username'] = user['username']
            flash('Logged in successfully!', 'success')
            return redirect(url_for('index'))
        else:
            flash('Invalid username or password', 'danger')
    
    return render_template('login.html')

@app.route('/logout')
def logout():
    session.clear()
    flash('You have been logged out', 'info')
    return redirect(url_for('index'))

@app.route('/upload', methods=['GET', 'POST'])
def upload():
    if 'user_id' not in session:
        flash('Please log in to upload files', 'warning')
        return redirect(url_for('login'))
    
    if request.method == 'POST':
        title = request.form['title']
        description = request.form.get('description', '')
        file = request.files['file']
        password = request.form['password']
        
        if not title or not file or not password:
            flash('Title, file, and password are required', 'danger')
            return redirect(url_for('upload'))
        
        if file:
            # Save original file
            filename = secure_filename(file.filename)
            file_path = os.path.join(app.config['UPLOAD_FOLDER'], filename)
            file.save(file_path)
            
            # Encrypt file
            encrypted_filename = f"enc_{os.urandom(8).hex()}_{filename}"
            encrypted_path = os.path.join(app.config['UPLOAD_FOLDER'], encrypted_filename)
            encrypt_file(file_path, encrypted_path, password)
            
            # Get file size
            file_size = os.path.getsize(encrypted_path)
            
            # Save to database
            conn = get_db()
            cursor = conn.cursor()
            cursor.execute('''INSERT INTO torrents 
                           (title, description, filename, encrypted_filename, size, uploader_id, decryption_key) 
                           VALUES (?, ?, ?, ?, ?, ?, ?)''', 
                           (title, description, filename, encrypted_filename, file_size, 
                            session['user_id'], password))
            conn.commit()
            conn.close()
            
            # Remove original
            os.remove(file_path)
            
            flash('File uploaded and encrypted successfully!', 'success')
            return redirect(url_for('index'))
    
    return render_template('upload.html')

@app.route('/download/<int:torrent_id>')
def download(torrent_id):
    conn = get_db()
    cursor = conn.cursor()
    cursor.execute('''SELECT * FROM torrents WHERE id = ?''', (torrent_id,))
    torrent = cursor.fetchone()
    conn.close()
    
    if not torrent:
        flash('Torrent not found', 'danger')
        return redirect(url_for('index'))
    
    # Get decryption key
    password = torrent['decryption_key']
    
    # Paths
    encrypted_path = os.path.join(app.config['UPLOAD_FOLDER'], torrent['encrypted_filename'])
    decrypted_filename = torrent['filename']
    decrypted_path = os.path.join(app.config['UPLOAD_FOLDER'], f"dec_{os.urandom(8).hex()}_{decrypted_filename}")
    
    # Decrypt
    decrypt_file(encrypted_path, decrypted_path, password)
    
    # Send file
    response = send_file(
        decrypted_path,
        as_attachment=True,
        download_name=decrypted_filename
    )
    
    # Schedule deletion
    response.call_on_close(lambda: os.remove(decrypted_path))
    
    return response

@app.route('/profile')

    cursor.execute('''SELECT * FROM torrents WHERE uploader_id = ?''', (session['user_id'],))
    user_torrents = cursor.fetchall()
    conn.close()
    
    return render_template('profile.html', user=user, torrents=user_torrents)
